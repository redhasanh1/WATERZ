import requests

# Initialize API client
API_KEY = "your_api_key_here"
BASE_URL = "https://api.markremoverai.com/v1"

# Upload video and create removal job
with open("input.mp4", "rb") as video, open("mask.png", "rb") as mask:
    response = requests.post(
        f"{BASE_URL}/jobs",
        headers={"Authorization": f"Bearer {API_KEY}"},
        files={
            "video": video,
            "mask": mask
        },
        data={"webhook_url": "https://your-app.com/webhook"}
    )

job = response.json()
print(f"Job {job['job_id']} created - Status: {job['status']}")
// Using fetch API
const API_KEY = 'your_api_key_here';

async function removeObject(videoFile, maskFile) {
  const formData = new FormData();
  formData.append('video', videoFile);
  formData.append('mask', maskFile);
  formData.append('webhook_url', 'https://your-app.com/webhook');

  const response = await fetch('https://api.markremoverai.com/v1/jobs', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${API_KEY}`
    },
    body: formData
  });

  const job = await response.json();
  console.log(`Job ${job.job_id} created`);
  return job;
}
